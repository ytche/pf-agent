### R-S1：5 例 save.effect 污染判定

```bash
python3 -c "
import json
IDS = ['spell_药剂与毒药P&P_法术_0005','spell_药剂与毒药P&P_法术_0006',
       'spell_异能选集PA_法术_0014','spell_异能选集PA_法术_0015','spell_page_702_0011']
with open('vectorizer/output/法术/chunks.jsonl') as f:
    chunks = {json.loads(l).get('chunk_id'): json.loads(l) for l in f}
dirty = 0
for cid in IDS:
    c = chunks.get(cid)
    if not c: print('  !! chunk 不存在:', cid); continue
    e = str((c['metadata'].get('saving_throw') or {}).get('effect') or '')
    polluted = '。' in e or '抗力' in e or len(e) > 45
    dirty += polluted
    print('%-36s len=%-4d polluted=%s  %r' % (cid, len(e), polluted, e[:50]))
print('污染残留:', dirty, '(路径A 通过标准=0；路径B 不跑本脚本，审 KN 与文档)')
"
```

### R-S2：R2 三项 KN 的规模数字实测

```bash
python3 -c "
import json
with open('vectorizer/output/法术/chunks.jsonl') as f:
    chunks = [json.loads(l) for l in f]
spell = [c for c in chunks if c.get('metadata',{}).get('field_status')]

# KN012: text 含 **法术抗力：** 有/可 但 sr 未 parsed
kn012 = 0
for c in spell:
    t = c['text']; i = t.find('**法术抗力：**')
    if i < 0 or c['metadata']['field_status'].get('spell_resistance') == 'parsed': continue
    val = t[i+8: i+30].lstrip('*： ')
    if val.startswith(('有','可')) and not val.startswith('不可'): kn012 += 1

# KN013: 字符串字段 parsed 值吞正文
kn013 = {}
for f in ['duration','target','effect','school']:
    parsed = [c for c in spell if c['metadata']['field_status'].get(f)=='parsed']
    limit = 15 if f=='school' else 40
    kn013[f] = (sum(1 for c in parsed if '。' in str(c['metadata'].get(f) or '')),
                sum(1 for c in parsed if len(str(c['metadata'].get(f) or ''))>limit))

# KN014: domains∩subdomains
kn014 = sum(1 for c in spell
            if {d['name'] for d in (c['metadata'].get('domains') or [])} &
               {s['name'] for s in (c['metadata'].get('subdomains') or [])})

print('KN012 (sr 真值丢失):', kn012, ' 基线 395')
print('KN013 (含句号/超长):', kn013, ' 基线 duration(179,162) target(31,72) effect(24,44) school(11,25)')
print('KN014 (domains∩subdomains):', kn014, ' 基线 125')
"
```

注：R2 不修代码，三项数字应与基线**完全一致**；若不一致说明 pipeline 被重跑过或代码被改动，需实施者说明。
