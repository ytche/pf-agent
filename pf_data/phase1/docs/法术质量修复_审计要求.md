```bash
python3 -c "
import json
with open('vectorizer/output/法术/chunks.jsonl') as f:
    chunks = [json.loads(l) for l in f]
spell = [c for c in chunks if c.get('metadata',{}).get('field_status')]
sr = sum(1 for c in spell if c['metadata'].get('spell_resistance') and isinstance(c['metadata']['spell_resistance'],dict) and c['metadata']['spell_resistance'].get('note') and len(str(c['metadata']['spell_resistance']['note']))>20)
sv = sum(1 for c in spell if c['metadata'].get('saving_throw') and isinstance(c['metadata']['saving_throw'],dict) and c['metadata']['saving_throw'].get('effect') and ('**' in str(c['metadata']['saving_throw']['effect']) or len(str(c['metadata']['saving_throw']['effect']))>15))
print('sr.note>20:', sr, ' save.effect dirty:', sv)
"
```

### S2：F2 裸标签计数 + 分书覆盖率

```bash
python3 -c "
import json, re
with open('vectorizer/output/法术/chunks.jsonl') as f:
    chunks = [json.loads(l) for l in f]
for label, key in [('施放时间','casting_time'), ('持续','duration'), ('抗力','spell_resistance')]:
    pat = r'(?<!\*)' + label + '：'
    n = sum(1 for c in chunks if re.search(pat, c['text']) and c.get('metadata',{}).get('field_status',{}).get(key)!='parsed')
    print('裸标签「%s」未 parsed:' % label, n)
for book in ['COMP','MA','CRB','APG','UM','OA','UI']:
    sub = [c for c in chunks if c['book_abbreviation']==book and c.get('metadata',{}).get('field_status')]
    row = ['%s=%.0f%%' % (k, sum(1 for c in sub if c['metadata']['field_status'].get(k)=='parsed')/len(sub)*100) for k in ['school','spell_level','casting_time','range','duration','saving_throw']]
    print('%-6s n=%4d  %s' % (book, len(sub), '  '.join(row)))
"
```

### S4：field_status 诚实性交叉检查

```bash
python3 -c "
import json
def empty(v):
    if v is None or v=='' or v==[] or v=={}: return True
    if isinstance(v,dict): return all(empty(x) for x in v.values())
    return False
with open('vectorizer/output/法术/chunks.jsonl') as f:
    chunks = [json.loads(l) for l in f]
bad1 = bad2 = 0
for c in chunks:
    m = c.get('metadata',{})
    fs = m.get('field_status')
    if not fs: continue
    for k, st in fs.items():
        v = m.get(k)
        if st=='parsed' and empty(v): bad1 += 1
        if st=='missing' and not empty(v): bad2 += 1
print('parsed 但值为空:', bad1, ' missing 但值非空:', bad2)
"
```
